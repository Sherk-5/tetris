    usleep(500000);  // 0.5 sec   → ~2 FPS
    usleep(100000);  // 0.1 sec   → ~10 FPS
    usleep(50000);   // 0.05 sec  → ~20 FPS
    usleep(16000);   // 0.016 sec → ~60 FPS
    usleep(8000);    // 0.008 sec → ~125 FPS
